def totp(secret):
    s=secret.replace(' ','').upper(); s+='='*((8-len(s)%8)%8); key=base64.b32decode(s)
    c=struct.pack('>Q', int(time.time())//30); h=hmac.new(key,c,hashlib.sha1).digest(); o=h[19]&15
    return str((struct.unpack('>I',h[o:o+4])[0]&0x7fffffff)%10**6).zfill(6)
